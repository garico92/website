{{/* media-nav — a strip of small thumbnails linking to every `media-item` on the page, in page order. Usage: {{< media-nav >}} (anywhere above the items). Titles are read from the media-item calls; thumbnails from manifest.json. */}} {{- $man := dict -}} {{- with .Page.Resources.Get "manifest.json" -}}{{- $man = . | transform.Unmarshal -}}{{- end -}} {{- $res := .Page.Resources -}} {{- $calls := findRESubmatch `(?s)\{\{<\s*media-item\s+id="([^"]+)"\s+title="([^"]+)"` .Page.RawContent -}}
{{- range $calls }} {{- $id := index . 1 }}{{ $title := index . 2 }} {{- $it := index ($man.items | default dict) $id }} {{- $src := "" }} {{- with $it }}{{ with $res.Get (.poster | default .preview) }}{{ $src = (.Fill "320x180 Center webp q70").RelPermalink }}{{ end }}{{ end }} {{- with $src }}{{ end }} {{ $title }} {{- end }}
