{{/* photo-grid — a grid of photos from the page bundle; click one to see it large. Usage, one photo per line ("file | caption"): {{< photo-grid >}} featured.jpg | On stage at ARCI L'Impegno. poster.jpg | The poster. {{< /photo-grid >}} Styles: .ga-photos in assets/css/custom.css. Lightbox: native , no library. */}} {{- $page := .Page -}} {{- $items := slice -}} {{- range split (trim .Inner "\n ") "\n" -}} {{- $l := trim . " " -}} {{- if $l -}} {{- $parts := split $l "|" -}} {{- $f := trim (index $parts 0) " " -}} {{- $cap := "" }}{{ if gt (len $parts) 1 }}{{ $cap = trim (delimit (after 1 $parts) "|") " " }}{{ end -}} {{- with $page.Resources.Get $f -}}{{- $items = $items | append (dict "r" . "cap" $cap) -}}{{- else -}}{{- warnf "photo-grid: %s not found in %s" $f $page.File.Path -}}{{- end -}} {{- end -}} {{- end -}} {{- $id := printf "pg-%d" .Ordinal -}}
