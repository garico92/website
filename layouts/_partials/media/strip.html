{{/* media/strip — the videos of the Media page as small cards that play in place (same player as the talk gallery). Used by the `media-strip` shortcode (talks page) and the `media-teaser` home-page block. Input dict: page the page being rendered (registers the player JS) mediapage optional, path of the media page (default "/media") ids optional, comma-separated media-item ids to show, in this order (default: all videos) more optional, text of the link to the Media page under the strip hide_more optional, true hides that link Titles and badges are read from the media-item calls of the media page, files from its manifest.json. */}} {{- $mp := site.GetPage (.mediapage | default "/media") -}} {{- if not $mp }}{{ errorf "media-strip: media page not found" }}{{ end -}} {{- .page.Store.Set "talks_js" true -}} {{- $man := dict -}} {{- with $mp.Resources.Get "manifest.json" }}{{ $man = . | transform.Unmarshal }}{{ end -}} {{- $res := $mp.Resources -}} {{- $calls := findRESubmatch `(?s)\{\{<\s*media-item\s+id="([^"]+)"\s+title="([^"]+)"(.*?)>\}\}` $mp.RawContent -}} {{- $want := slice }}{{ with .ids }}{{ range split . "," }}{{ $want = $want | append (trim . " ") }}{{ end }}{{ end -}} {{- $more := .more | default "Descriptions, downloads and the deep-zoom picture on the Media page →" -}} {{- $show_more := not .hide_more -}} {{- $cards := slice -}} {{- range $calls -}} {{- $id := index . 1 }}{{ $it := index ($man.items | default dict) $id -}} {{- if or (not $it) (eq $it.kind "image") }}{{ continue }}{{ end -}} {{- if and $want (not (in $want $id)) }}{{ continue }}{{ end -}} {{- $badge := "" }}{{ with findRESubmatch `badge="([^"]+)"` (index . 3) 1 }}{{ $badge = index (index . 0) 1 }}{{ end -}} {{- $cards = $cards | append (dict "id" $id "title" (index . 2) "it" $it "badge" $badge) -}} {{- end -}} {{- if $want -}} {{- $ordered := slice -}} {{- range $want }}{{ $w := . }}{{ range $cards }}{{ if eq .id $w }}{{ $ordered = $ordered | append . }}{{ end }}{{ end }}{{ end -}} {{- $cards = $ordered -}} {{- end -}}
{{- range $cards }} {{- $it := .it -}} {{- $src := "" }}{{ $src_sm := "" -}} {{- if eq $it.kind "loop" }}{{ with $res.Get $it.file }}{{ $src = .RelPermalink }}{{ end -}} {{- else -}} {{- with $it.renditions -}} {{- range $k, $r := . }}{{ with $res.Get $r.file }}{{ if or (eq $k "default") (eq $k "1080p") }}{{ $src = .RelPermalink }}{{ else if eq $k "720p" }}{{ $src_sm = .RelPermalink }}{{ end }}{{ end }}{{ end -}} {{- end -}} {{- end -}} {{- $poster := "" }}{{ with $res.Get $it.poster }}{{ $poster = . }}{{ end -}} {{- $dur := int (math.Round $it.duration_s) -}} {{- $link := printf "%s#%s" $mp.RelPermalink (.id | anchorize) -}}
